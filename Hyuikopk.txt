WhatTheHeck=''
entity:SetCallback("OnSpawned", function()
	SpawningSound=writefile("SpawningDima.wav", game:HttpGet("https://github.com/therealvan45/Doors-ShizoMode/raw/refs/heads/main/hyuikopkSpawn.wav"))
    local SpwnSnd=Instance.new("Sound",game:service("Workspace"))
    SpwnSnd.SoundId=getcustomasset("SpawningDima.wav")
    SpwnSnd.Volume=4
    SpwnSnd:Play()
    SpwnSnd.Ended:Connect(function()
        SpwnSnd:Destroy()
    end)

    WhatTheHeck=coroutine.wrap(function()
        ThisStupidGuy=entity.Model
        Torsopart=ThisStupidGuy.HumanoidRootPart
        plr=game.Players.LocalPlayer


        local function spit(Appearence)
            task.spawn(function()
                local SpitBall=Instance.new("Part",workspace)
                SpitBall.Position=Appearence
                SpitBall.Size=Vector3.new(2,2,2)
                SpitBall.CanCollide=false
                SpitBall.BrickColor=BrickColor.new("White")
                SpitBall.Shape=Enum.PartType.Ball
                SpitBall.Name="Spitty"
                SpitBall.CFrame=CFrame.new(SpitBall.Position, plr.Character.HumanoidRootPart.Position)
                SpitBall.AssemblyLinearVelocity=SpitBall.CFrame.LookVector *(plr.Character.HumanoidRootPart.Position - Torsopart.Position).Magnitude * (2/0.5) +Vector3.new(0,20,0)
                local SpitSound=Instance.new("Sound",SpitBall) :: Sound
                SpitSound.Volume=5
                SpitSound.RollOffMaxDistance=30
                SpitSound.RollOffMinDistance=1
                SpitSound.SoundId="rbxassetid://350906125"
                SpitSound:Play()

                SpitBall.Touched:Connect(function(hit)
                    if hit.Parent:FindFirstChildOfClass("Humanoid") and hit.Parent~=ThisStupidGuy then
                        hit.Parent:FindFirstChildOfClass("Humanoid").Health-=5
                        SpitBall:Destroy()
                    end
                end)

                
            end)
        end
        SixSeven = task.spawn(function()
            while true do
                wait()
                ThisStupidGuy.RightUpperArm.RightElbowRigAttachment.CFrame=CFrame.new(0.001, -0.471, 0) * CFrame.Angles(math.rad(90 + math.sin(tick()*24)*30),0,0)
                ThisStupidGuy.LeftUpperArm.LeftElbowRigAttachment.CFrame=CFrame.new(-0, -0.471, 0) * CFrame.Angles(math.rad(90 + math.sin(tick()*24)*-30),0,0)
            end
        end)
        while true do
            
            if (plr.Character.HumanoidRootPart.Position - Torsopart.Position).Magnitude <= 30 then
                spit(ThisStupidGuy.Head.Position)
                
            end
            wait(0.3)
        end
    end)
    WhatTheHeck()
end)
