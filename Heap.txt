---====== Debug entity ======---
local function SomeStupidSoundPlay(Aud,Parent,vlm,min,max,loop)
    local aud=Instance.new("Sound",Parent)
    aud.Volume=vlm
    aud.SoundId=Aud
    aud.RollOffMaxDistance=max
    aud.RollOffMinDistance=min
    aud.Looped=loop
    aud:Play()
    aud.Ended:Once(function()
        aud:Destroy()
    end)
        
end
